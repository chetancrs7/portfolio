```python title="src/inference/predict.py" showLineNumbers {4-7}
async def predict_sentiment(text: str, model: FusionModel) -> Prediction:
    semantic_vector = await semantic_encoder.embed(text)
    structural_features = structural_extractor.transform(text)

    prediction = model.fuse(
        semantic_vector=semantic_vector,
        structural_features=structural_features,
    )
    return prediction
```

```tsx title="src/app/work/[slug]/page.tsx" showLineNumbers {5}
  const { slug } = await params;
  const item = getWorkBySlug(slug);

  if (!item) notFound();

  const content = await getWorkContent(item);
  return content?.Content ? <content.Content /> : <CaseStudyFallback />;
}
```
